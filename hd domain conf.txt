sudo nano /etc/nginx/sites-available/sku.io.vn.conf


server {
    listen 80;
    listen [::]:80;

    server_name sku.io.vn www.sku.io.vn;

    root /var/www/wp_sku;
    index index.php index.html;

    charset utf-8;

    # Logs
    access_log /var/log/nginx/sku.access.log;
    error_log /var/log/nginx/sku.error.log warn;

    # Upload limit
    client_max_body_size 64M;

    # WordPress permalinks
    location / {
        try_files $uri $uri/ /index.php?$args;
    }

    # PHP 8.2-FPM
    location ~ \.php$ {
        include snippets/fastcgi-php.conf;
        fastcgi_pass unix:/run/php/php8.2-fpm.sock;

        fastcgi_read_timeout 300;
    }

    # Cache static assets
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|webp|woff|woff2|ttf|eot)$ {
        expires 30d;
        add_header Cache-Control "public";
        access_log off;
        try_files $uri =404;
    }

    # Block hidden files, except ACME challenge
    location ~ /\.(?!well-known/) {
        deny all;
    }

    # Protect sensitive WordPress files
    location ~* ^/(wp-config\.php|readme\.html|license\.txt)$ {
        deny all;
    }

    # Block sensitive files
    location ~* \.(env|ini|log|sql|bak|backup|old|swp)$ {
        deny all;
    }

    # Block PHP execution inside uploads
    location ~* ^/wp-content/uploads/.*\.php$ {
        deny all;
    }

    # Block access to XML-RPC (optional)
    # Uncomment if XML-RPC is not needed
    # location = /xmlrpc.php {
    #     deny all;
    # }
}

--

sudo ln -s /etc/nginx/sites-available/sku.io.vn.conf /etc/nginx/sites-enabled/sku.io.vn.conf

sudo nginx -t && sudo systemctl reload nginx

sudo certbot --nginx \
-d sku.io.vn \
-d www.sku.io.vn
